CREATE OR REPLACE PROCEDURE `seaware.plan_forecast_actuals_sp`()
begin

declare cur_metric_vals ARRAY<STRING>;
declare sql_column_str string default '';
declare sql_select_str string default '';
declare sql_case_str string default '';
declare sql_create_ddl string;
declare sql string;
declare sql_str1 string;
declare sql_str2 string;
declare sql_str3 string;
DECLARE sql_drop_ddl string;

set sql_str1= 'insert into seaware.plan_forecast_actuals_rpt WITH max_snapshot_date AS ( SELECT sail.src_sail_id src_sail_id, MAX(snapshot_date) snapshot_date, Date_diff(Date_add(cast(sail_date_from as date),interval -1 day),Date_add(snapshot_date,interval -1 day),week) week_index FROM hive_schema_stg.seaware_revenue_fact fact JOIN seaware.seaware_sail_dim sail ON fact.sail_id=sail.sail_id GROUP BY sail.src_sail_id, week_index ) SELECT DISTINCT plan_forecast.cabin_ctg_generic, plan_forecast.package_type, plan_forecast.src_sail_id, actuals.sail_days, plan_forecast.week_index, plan_forecast.sunday_date, plan_forecast.forecast_version_desc,';

set sql_str2= ' actuals.Actual_Gross_Onboard_Revenue, actuals.Actual_Gross_Onboard_Revenue_USD, actuals.actual_gross_other_revenue, actuals.actual_gross_other_revenue_usd, actuals.actual_gross_ticket_revenue, actuals.actual_gross_ticket_revenue_usd, actuals.actual_gross_total_revenue, actuals.actual_gross_total_revenue_usd, actuals.actual_other_gpd_usd, actuals.actual_ticket_gpd_usd, actuals.actual_total_gpd_usd, actuals.Net_Ticket_Revenue, actuals.Net_Ticket_Revenue_USD, actuals.Net_APD, actuals.Net_APD_USD, actuals.Guest_Count, cabin_fact.cabin_occ, cabin_fact.ok_abs, cabin_fact.gty_abs, cabin_fact.ok_gtry_rstr_abs, cabin_fact.num_of_cabins, cabin_fact.na_abs, cabin_fact.ok_wgt, cabin_fact.gty_wgt, cabin_fact.wtl_abs, cabin_fact.wtl_wgt, cabin_fact.avail_abs, cabin_fact.avail_wgt, cabin_fact.num_of_reserved FROM ( SELECT seaware_plan_forecast_header_dim.cabin_ctg_generic AS cabin_ctg_generic, seaware_plan_forecast_header_dim.package_type AS package_type, seaware_sail_dim.src_sail_id AS src_sail_id, seaware_plan_forecast_value_dim.week_index AS week_index, CASE WHEN (EXTRACT (DAYOFWEEK FROM (date_add(cast(seaware_sail_dim.sail_date_from as date), INTERVAL seaware_plan_forecast_value_dim.week_index week ))))=7 THEN (date_add(cast(seaware_sail_dim.sail_date_from as date), INTERVAL seaware_plan_forecast_value_dim.week_index week )) ELSE DATE_TRUNC((date_add(date_add(cast(seaware_sail_dim.sail_date_from as date), INTERVAL seaware_plan_forecast_value_dim.week_index week ), INTERVAL 7 day)),week(sunday)) END AS sunday_date, seaware_forecastversion_lkp.forecast_version_desc AS forecast_version_desc ';

set sql_str3= ' FROM seaware.seaware_plan_forecast_header_dim seaware_plan_forecast_header_dim INNER JOIN seaware.seaware_plan_forecast_value_dim seaware_plan_forecast_value_dim ON (seaware_plan_forecast_header_dim.plan_forecast_id = seaware_plan_forecast_value_dim.plan_forecast_id) INNER JOIN seaware.sw_forecastmetric_lkp seaware_forecastmetric_lkp ON (seaware_plan_forecast_header_dim.forecast_metric_id = seaware_forecastmetric_lkp.forecast_metric_id) INNER JOIN seaware.sw_forecastversion_lkp seaware_forecastversion_lkp ON (seaware_plan_forecast_header_dim.forecast_version_id = seaware_forecastversion_lkp.forecast_version_id) INNER JOIN seaware.seaware_sail_dim seaware_sail_dim ON (seaware_plan_forecast_header_dim.sail_id = seaware_sail_dim.sail_id) GROUP BY cabin_ctg_generic, package_type, src_sail_id, week_index, forecast_version_desc, sail_date_from ORDER BY src_sail_id, week_index ) AS plan_forecast LEFT JOIN ( SELECT generic_category, package_type, tmp.src_sail_id src_sail_id, sail_days, tmp.snapshot_date snapshot_date, tmp.week_index week_index, forecast_version_desc, forecast_metric_desc, SUM(actual_gross_onboard_revenue) actual_gross_onboard_revenue, SUM(actual_gross_onboard_revenue_usd) actual_gross_onboard_revenue_usd, SUM(actual_gross_other_revenue) actual_gross_other_revenue, SUM(actual_gross_other_revenue_usd) actual_gross_other_revenue_usd, SUM(actual_gross_ticket_revenue) actual_gross_ticket_revenue, SUM(actual_gross_ticket_revenue_usd) actual_gross_ticket_revenue_usd, SUM(actual_gross_total_revenue) actual_gross_total_revenue, SUM(actual_gross_total_revenue_usd) actual_gross_total_revenue_usd, SUM(actual_other_gpd_usd) actual_other_gpd_usd, SUM(actual_ticket_gpd_usd) actual_ticket_gpd_usd, SUM(actual_total_gpd_usd) actual_total_gpd_usd, SUM(net_ticket_revenue) net_ticket_revenue, SUM(net_ticket_revenue_usd) net_ticket_revenue_usd, SUM(net_apd) net_apd, SUM(net_apd_usd) net_apd_usd, SUM(guest_count) guest_count FROM ( SELECT cab_res.generic_category, pkg.package_type, sail.src_sail_id, sail.sail_days, snapshot_date, Date_diff( Date_add( cast(sail_date_from as date),Interval -1 DAY), Date_add( snapshot_date,interval -1 DAY),WEEK) week_index, \'Actual\' as forecast_version_desc, \'Actual\' as forecast_metric_desc, SUM(actual_gross_onboard_revenue) AS actual_gross_onboard_revenue, SUM(actual_gross_onboard_revenue_usd) AS actual_gross_onboard_revenue_usd, SUM(actual_gross_other_revenue) actual_gross_other_revenue, SUM(actual_gross_other_revenue_usd) actual_gross_other_revenue_usd, SUM(actual_gross_ticket_revenue) actual_gross_ticket_revenue, SUM(actual_gross_ticket_revenue_usd) actual_gross_ticket_revenue_usd, SUM(actual_gross_total_revenue) actual_gross_total_revenue, SUM(actual_gross_total_revenue_usd) actual_gross_total_revenue_usd, SUM(actual_gross_other_revenue_usd) /sail.sail_days actual_other_gpd_usd, SUM(actual_gross_ticket_revenue_usd)/sail.sail_days actual_ticket_gpd_usd, SUM(actual_gross_total_revenue_usd) /sail.sail_days actual_total_gpd_usd, SUM(CASE WHEN it.invoice_item_type IN (\'VOYAGE FARE\', \'MANUAL ADJ\') THEN amount ELSE 0 END ) AS net_ticket_revenue, SUM(CASE WHEN it.invoice_item_type IN (\'VOYAGE FARE\', \'MANUAL ADJ\') THEN amount_usd ELSE 0 END ) AS net_ticket_revenue_usd, SUM(CASE WHEN it.invoice_item_type IN (\'VOYAGE FARE\', \'MANUAL ADJ\') THEN amount ELSE 0 END ) /sail.sail_days AS net_apd, SUM(CASE WHEN it.invoice_item_type IN (\'VOYAGE FARE\', \'MANUAL ADJ\') THEN amount_usd ELSE 0 END ) /sail.sail_days AS net_apd_usd, COUNT(DISTINCT rev.guest_id) AS guest_count FROM seaware.seaware_revenue_fact rev INNER JOIN seaware.seaware_sail_dim sail ON (rev.sail_id = sail.sail_id) INNER JOIN seaware.seaware_package_dim pkg ON (rev.package_id = pkg.package_id) INNER JOIN ( SELECT generic_category, bcrd.res_id, bcrd.guest_id FROM seaware.seaware_booked_cabin_reservation_dim bcrd JOIN ( SELECT res_id, guest_id, MAX(rec_end_dttm) rec_end_dttm FROM seaware.seaware_booked_cabin_reservation_dim GROUP BY res_id, guest_id) bcrd_m ON bcrd.res_id=bcrd_m.res_id AND bcrd.guest_id=bcrd_m.guest_id AND bcrd.rec_end_dttm=bcrd_m.rec_end_dttm) AS cab_res ON (rev.res_id = cab_res.res_id AND rev.guest_id = cab_res.guest_id) INNER JOIN seaware.seaware_invoice_item_type_lkp it ON (rev.invoice_item_type_id = it.invoice_item_type_id) GROUP BY generic_category, package_type, sail.src_sail_id, sail.sail_date_from, snapshot_date, sail_days HAVING week_index > -111 AND week_index < 1) tmp INNER JOIN max_snapshot_date msd ON (tmp.src_sail_id=msd.src_sail_id AND tmp.week_index=msd.week_index AND tmp.snapshot_date=msd.snapshot_date) GROUP BY generic_category, package_type, tmp.src_sail_id, sail_days, tmp.snapshot_date, tmp.week_index, forecast_version_desc, forecast_metric_desc) AS actuals ON plan_forecast.src_sail_id = actuals.src_sail_id AND plan_forecast.cabin_ctg_generic = actuals.generic_category AND plan_forecast.package_type = actuals.package_type AND plan_forecast.week_index = actuals.week_index LEFT JOIN ( SELECT generic_category, package_type, tmp.src_sail_id src_sail_id, tmp.snapshot_date snapshot_date, tmp.week_index week_index, ok_abs, gty_abs, ok_gtry_rstr_abs, num_of_cabins, na_abs, ok_wgt, gty_wgt, wtl_abs, wtl_wgt, avail_abs, avail_wgt, num_of_reserved, cabin_occ FROM ( SELECT cab_fct.generic_category, pkg.package_type, sail.src_sail_id,  cast(snapshot_time as date)  snapshot_date, Date_diff( Date_add( cast(sail_date_from as date),interval -1 day), Date_add(  cast(snapshot_time as date),interval -1 day),week) week_index, SUM(coalesce(ok_abs, 0)) ok_abs, SUM(coalesce(gty_abs, 0)) gty_abs, SUM(coalesce(ok_gtry_rstr_abs, 0)) ok_gtry_rstr_abs, SUM(coalesce(num_of_cabins, 0)) num_of_cabins, SUM(coalesce(na_abs, 0)) na_abs, SUM(coalesce(ok_wgt, 0)) ok_wgt, SUM(coalesce(gty_wgt, 0)) gty_wgt, SUM(coalesce(wtl_abs, 0)) wtl_abs, SUM(coalesce(wtl_wgt, 0)) wtl_wgt, SUM(coalesce(avail_abs, 0)) avail_abs, SUM(coalesce(avail_wgt, 0)) avail_wgt, SUM(coalesce(num_of_reserved, 0)) num_of_reserved, CASE WHEN ( SUM(coalesce(num_of_cabins, 0))-SUM(coalesce(na_abs, 0)))=0 THEN 0 ELSE ((SUM(coalesce(ok_abs, 0)) + SUM(coalesce(gty_abs, 0)) + SUM(coalesce(ok_gtry_rstr_abs, 0))) / (SUM(coalesce(num_of_cabins, 0))-SUM(coalesce(na_abs, 0))))*100 END AS cabin_occ FROM hive_schema_stg.seaware_sail_cabin_fact cab_fct INNER JOIN seaware.seaware_sail_dim sail ON ( cab_fct.sail_id = sail.sail_id) INNER JOIN seaware.seaware_package_dim pkg ON ( sail.src_sail_id = pkg.src_sail_id) GROUP BY cab_fct.generic_category, pkg.package_type, sail.src_sail_id, snapshot_date, week_index ) tmp INNER JOIN max_snapshot_date msd ON (tmp.src_sail_id=msd.src_sail_id AND tmp.week_index=msd.week_index AND tmp.snapshot_date=msd.snapshot_date) ) AS cabin_fact ON plan_forecast.src_sail_id = cabin_fact.src_sail_id AND plan_forecast.cabin_ctg_generic = cabin_fact.generic_category AND plan_forecast.package_type = cabin_fact.package_type AND plan_forecast.week_index = cabin_fact.week_index;';

set cur_metric_vals = ARRAY(SELECT distinct forecast_metric_desc from hive_schema_stg.sw_forecastmetric_lkp);

for metric_lkp in (Select * from unnest(cur_metric_vals) as forecast_metric_desc)
DO
set sql_column_str = sql_column_str ||  replace(replace(replace(replace(metric_lkp.forecast_metric_desc, ' ', '_'),'%','PER'),'$','DOLLAR'),'-','_') || ' float64,';
SET sql_select_str = sql_select_str || ' plan_forecast.' || replace(replace(replace(replace(metric_lkp.forecast_metric_desc, ' ', '_'),'%','PER'),'$','DOLLAR'),'-','_')  || ',';
set sql_case_str = sql_case_str || ',' || 'sum(case when seaware_forecastmetric_lkp.forecast_metric_desc = \'' || replace(replace(replace(replace(metric_lkp.forecast_metric_desc, ' ', '_'),'%','PER'),'$','DOLLAR'),'-','_') || '\' then value else 0 end) as ' || replace(replace(replace(replace(metric_lkp.forecast_metric_desc, ' ', '_'),'%','PER'),'$','DOLLAR'),'-','_') ; 

END FOR;

#SET sql_drop_ddl = 'DROP TABLE IF EXISTS seaware.plan_forecast_actuals_rpt;';
SET sql_create_ddl = 'CREATE OR REPLACE TABLE  seaware.plan_forecast_actuals_rpt( cabin_ctg_generic string, package_type string, src_sail_id int, sail_days int, week_index int, sunday_date date, forecast_version_desc string,' || sql_column_str || 'actual_gross_onboard_revenue float64, actual_gross_onboard_revenue_usd float64, actual_gross_other_revenue float64, actual_gross_other_revenue_usd float64, actual_gross_ticket_revenue float64, actual_gross_ticket_revenue_usd float64, actual_gross_total_revenue float64, actual_gross_total_revenue_usd float64, actual_other_gpd_usd float64, actual_ticket_gpd_usd float64, actual_total_gpd_usd float64, net_ticket_revenue float64, net_ticket_revenue_usd float64, net_apd float64, net_apd_usd float64, guest_count int, cabin_occ float64, ok_abs float64, gty_abs float64, ok_gtry_rstr_abs float64, num_of_cabins float64, na_abs float64, ok_wgt float64, gty_wgt float64, wtl_abs float64, wtl_wgt float64, avail_abs float64, avail_wgt float64, num_of_reserved float64 ) ';
set sql = sql_str1 || sql_select_str || sql_str2||sql_case_str||sql_str3;

EXECUTE IMMEDIATE  sql_create_ddl;
EXECUTE IMMEDIATE  sql;

#select sql;

#select sql_create_ddl;
end;
