CREATE OR REPLACE PROCEDURE `seaware.commission_payout_rpt_sp`()
BEGIN
delete from seaware.commission_payout_rpt where true;
 insert into seaware.commission_payout_rpt 
  SELECT 
 safe_cast(agn.seaware_agency_id__c as int)  as seaware_agency_id,
  agn.id AS crm_agency_id,
  agn.name AS agency_name,
  safe_cast(seaware_agent_id__c as int)   as seaware_agent_id,
  agt.id  AS crm_agent_id,
  agt.name AS first_mate_name,
  rd_curr.src_res_id AS booking_number,
  rd_curr.src_group_id  AS circle_number,
  gd.first_name || ' ' || gd.last_name AS sailor_1_name,
  rd_curr.res_status AS booking_status,
  CASE
    WHEN rd_curr.res_status = 'OF' THEN 'On Hold'
    WHEN rd_curr.res_status = 'BK' THEN 'Booked'
    WHEN rd_curr.res_status = 'CL' THEN 'Sailed'
    WHEN rd_curr.res_status = 'TM' THEN 'Sailed'
    WHEN rd_curr.res_status = 'CX' THEN 'Cancelled'
    WHEN rd_curr.res_status = 'CT' THEN 'Cancelled'
END
  booking_status_desc,
  rd_curr.res_init_date AS booked_date,
  sd.sail_date_from AS voyage_date,
  t.trans_status,
  t.trans_type,
  t.form_of_trans,
  t.is_dest_entity,
  t.trans_time_stamp,
  CASE
    WHEN t.is_dest_entity = 'N' THEN (t.amount_full_trans - t.amount_full_trans * 2)
  ELSE
  t.amount_full_trans
END
  AS trans_amount,
  rd_curr.currency,
  ar_ap_due_date payout_date,
  current_timestamp etl_ld_dt
FROM
  hive_schema_stg.seaware_transaction_evt_dim t
JOIN
  seaware.seaware_reservation_dim rd_curr
ON
  t.res_id = rd_curr.res_id
  AND rd_curr.rec_end_dttm = '9999-12-31'
LEFT JOIN
  seaware.seaware_sail_dim sd
ON
  rd_curr.src_sail_id = sd.src_sail_id
  AND sd.rec_end_dttm = '9999-12-31'
LEFT JOIN
  seaware.seaware_res_guest_rel rel
ON
  rd_curr.res_id = rel.res_id
  AND rel.guest_seqn = 1
LEFT JOIN
  seaware.seaware_guest_dim gd
ON
  rel.guest_id = gd.guest_id
LEFT JOIN
  crm.crm_agency_master agn
ON
  rd_curr.seaware_agency_id = safe_cast(agn.seaware_agency_id__c as int64)
  AND agn.rec_end_dttm = '9999-12-31'
LEFT JOIN
  crm.crm_agent_master agt
ON
  rd_curr.seaware_agent_id = safe_cast(agt.seaware_agent_id__c as int64)
  AND agt.rec_end_dttm = '9999-12-31'
WHERE
  t.trans_type = 'COMMISSION'
  AND t.trans_status = 'OK';
	END;
