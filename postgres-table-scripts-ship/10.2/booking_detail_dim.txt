	booking_detail_skey int8 NOT NULL,
	voyage_id text NULL,
	booking_id text NULL,
	booking_link_id text NULL,
	status text NULL,
	cancellation_reason text NULL,
	waiver_code text NULL,
	waiver_status text NULL,
	payment_status text NULL,
	reservation_number text NULL,
