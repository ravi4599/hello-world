CREATE TABLE hive_schema_stg.request_detail_dim (
	request_detail_skey int8 NOT NULL,
	voyage_id text NOT NULL,
	request_id text NULL,
	request_guid text NULL,
	request_description text NULL,
	request_priority int4 NULL,
	request_category_id int4 NULL,
	request_category_name text NULL,
	request_type_id int4 NULL,
	request_type_name text NULL,
	request_type_priority int4 NULL,
	load_dt timestamp NULL,
	upd_dt timestamp NULL,
	primaryhash text NOT NULL,
	md5_hash text NOT NULL
);


CREATE OR REPLACE VIEW shipdw.request_detail_dim
AS SELECT request_detail_dim.request_detail_skey,
    request_detail_dim.voyage_id,
    request_detail_dim.request_id,
    request_detail_dim.request_guid,
    request_detail_dim.request_description,
    request_detail_dim.request_priority,
    request_detail_dim.request_category_id,
    request_detail_dim.request_category_name,
    request_detail_dim.request_type_id,
    request_detail_dim.request_type_name,
    request_detail_dim.request_type_priority,
    request_detail_dim.load_dt,
    request_detail_dim.upd_dt,
    request_detail_dim.primaryhash,
    request_detail_dim.md5_hash
   FROM hive_schema_stg.request_detail_dim;
