CREATE TABLE hive_schema_stg.order_fact (
	voyage_id text NOT NULL,
	ship_code text NOT NULL,
	voyage_skey int8 NOT NULL,
	order_detail_skey int8 NOT NULL,
	venue_skey int8 NULL,
	order_placed_by_person_skey int8 NULL,
	order_delivered_to_person_skey int8 NULL,
	order_delivered_by_person_skey int8 NULL,
	sale_detail_skey int8 NULL,
	order_placed_datetime timestamp NULL,
	order_modified_datetime timestamp NULL,
	order_min_wait_time int4 NULL,
	order_max_wait_time int4 NULL,
	order_delivery_time_taken_mins int4 NULL,
	order_delayed_flg bool NULL,
	order_delivered_datetime timestamp NULL,
	order_amount float8 NULL,
	order_cancellation_date timestamp NULL,
	order_cancellation_refund_amount float8 NULL,
	order_cancellation_reason text NULL,
	order_total_number_items int4 NULL,
	order_origin_venue_skey int8 NULL,
	order_delivered_at_venue_skey int8 NULL,
	load_dt timestamp NOT NULL,
	upd_dt timestamp NOT NULL,
	primaryhash text NOT NULL,
	md5_hash text NOT NULL,
	part_dt text NOT NULL
);
CREATE INDEX idx_order_fact_type_categval ON hive_schema_stg.order_fact USING btree (ship_code, voyage_skey, order_detail_skey, venue_skey, order_placed_by_person_skey, order_delivered_to_person_skey, order_delivered_by_person_skey, sale_detail_skey,order_origin_venue_skey,order_delivered_at_venue_skey);

CREATE or replace VIEW shipdw.order_fact  AS SELECT * FROM hive_schema_stg.order_fact;
