CREATE OR REPLACE VIEW shipdw.restriction_dim
AS SELECT restriction_dim.voyage_id,
	restriction_dim.restriction_skey,	
	restriction_dim.person_id,
	restriction_dim.restrictions_id as restriction_id,
	restriction_dim.restrictions_guid as restriction_guid,
	restriction_dim.restrictions_type as restriction_type,
	restriction_dim.categoryid as restriction_categoryid,
	restriction_dim.category_code as restriction_category_code,
	restriction_dim.category as restriction_category,
	restriction_dim.comment as restriction_comment,
	restriction_dim.from_date as restriction_from_date,
	restriction_dim.to_date as restriction_to_date,
	restriction_dim.deleted_flag as src_deleted_flag,
	restriction_dim.created_date as src_created_date,
	restriction_dim.load_dt as load_dt,
	restriction_dim.upd_dt as upd_dt 
FROM hive_schema_stg.restriction_dim;
