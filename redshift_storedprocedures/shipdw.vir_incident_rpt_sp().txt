CREATE
	OR REPLACE PROCEDURE shipdw.vir_incident_rpt_sp () LANGUAGE plpgsql AS $$
BEGIN
	DELETE
	FROM shipdw.ship_vir_incident_rpt;

	INSERT INTO shipdw.ship_vir_incident_rpt (
		SELECT CAST("Incident_Details"."incident_id" AS TEXT) AS "incident_id"
		,CAST("Incident_Category"."incident_category_types" AS TEXT) AS "incident_type"
		,CAST("Incident_Category"."incident_parent_category_name" AS TEXT) AS "incident_category"
		,CAST("Incident_Details"."incident_reference_id" AS TEXT) AS "VIR_number"
		,Incident_Details."incident_priority" AS "priority"
		,Incident_Details."incident_severity" AS "severity"
		,Incident_Measures."incident_status" AS "status"
		,Incident_Measures."incident_created_datetime" AS "incident_opened_datetime"
		,Incident_Measures."incident_created_datetime_utc" AS "incident_opened_datetime_utc"
		,Incident_Measures."resolution_time" AS "total_resolution_time"
		,CAST("Reported_By"."person_first_name" AS TEXT) || ' ' || CAST("Reported_By"."person_last_name" AS TEXT) AS "reported_by"
		,Reported_By.seaware_id AS "reported_by_id"
		,CAST("Assigned_Crew"."person_first_name" AS TEXT) || ' ' || CAST("Assigned_Crew"."person_last_name" AS TEXT) AS "assigned_crew"
		,Assigned_Crew.seaware_id AS "assigned_crew_id"
		,CAST("Impacted_Guest"."person_first_name" AS TEXT) || ' ' || CAST("Impacted_Guest"."person_last_name" AS TEXT) AS "impacted_sailor"
		,Impacted_Guest.seaware_id AS "impacted_sailor_id"
		,CAST("Incident_Details"."incident_raised_by_notes" AS TEXT) AS "sailor_notes"
		,CAST("Incident_Details"."incident_attended_by_notes" AS TEXT) AS "crew_notes"
		,Incident_Details."incident_resolution_amount" AS "incident_resolution_amount"
		,Incident_Category."incident_category_dept_code" AS "incident_department"
		,Cabin."cabin_number" AS "cabin_number"
		,Voyage.voyage_number AS "voyage_number"
		,Voyage.voyage_description AS "voyage_description"
		,Voyage.ship_name AS "ship_name"
		,Current_timestamp AS etl_ld_dt 
		--WARNING! ERRORS ENCOUNTERED DURING SQL PARSING!
FROM "shipdw"."incident_fact" "Incident_Measures"
LEFT JOIN "shipdw"."incident_detail_dim" "Incident_Details" ON ("Incident_Measures"."incident_detail_skey" = "Incident_Details"."incident_detail_skey")
LEFT JOIN "shipdw"."incident_category_dim" "Incident_Category" ON ("Incident_Measures"."incident_category_code_skey" = "Incident_Category"."incident_category_skey")
LEFT JOIN "shipdw"."voyage_dim" "Voyage" ON ("Incident_Measures"."voyage_skey" = "Voyage"."voyage_skey")
LEFT JOIN "shipdw"."person_dim" "Assigned_Crew" ON (CAST(TRUNC("Incident_Measures"."assigned_to_person_skey") AS BIGINT) = "Assigned_Crew"."person_skey") /*LEFT JOIN "shipdw"."person_dim" "Duty_Manager" ON (CAST(TRUNC("Incident_Measures"."duty_manager_person_skey") AS BIGINT) = "Duty_Manager"."person_skey")*/
LEFT JOIN "shipdw"."person_dim" "Impacted_Guest" ON (CAST(TRUNC("Incident_Measures"."impacted_guest_person_skey") AS BIGINT) = "Impacted_Guest"."person_skey")
LEFT JOIN "shipdw"."person_dim" "Reported_By" ON (CAST(TRUNC("Incident_Measures"."reported_by_person_skey") AS BIGINT) = "Reported_By"."person_skey") /*LEFT JOIN "shipdw"."date_dim" "Due_Date" ON ("Incident_Measures"."due_date_skey" = "Due_Date"."date_id") LEFT JOIN "shipdw"."time_dim" "Due_Time" ON ("Incident_Measures"."due_time_skey" = "Due_Time"."time_skey")*/
LEFT JOIN "shipdw"."cabin_dim" "Cabin" ON ("Incident_Measures"."cabin_skey" = "Cabin"."cabin_skey") /*LEFT JOIN "shipdw"."venue_dim" "Venue" ON ("Incident_Measures"."venue_skey" = "Venue"."venue_skey") LEFT JOIN "shipdw"."date_dim" "Incident_Created" ON (CAST("Incident_Measures"."incident_created_datetime" AS DATE) = "Incident_Created"."date")*/
	);END;$$;
