
create or replace view seaware.seaware_revenue_fact
as
SELECT snapshot_date,RES_ID,GUEST_ID,package_id,ship_id,agency_id,sail_id,invoice_item_type_id,price_area_id,addon_id,promotion_id,
Amount,
VOYAGE_AMOUNT,TAXESandFEES,
NVL(MANUAL_ADJ,0) AS MANUAL_ADJ,
NVL(VOYAGE_PROTECT,0) AS INSURANCE,
((VOYAGE_AMOUNT+NVL(MANUAL_ADJ,0))* COMMISSION_PERCENT )/100 AS VOYAGE_COMMISSION_AMT,
(TAXESandFEES *COMMISSION_PERCENT )/ 100 AS TAX_COMMISION_AMT,
nvl(VOYAGE_AMOUNT*0.025,0) AS CREDIT_CARD_FEE,
(nvl(VOYAGE_AMOUNT,0)+nvl(TAXESandFEES,0)+nvl(MANUAL_ADJ,0)) AS ACTUAL_GROSS_TICKET_REVENUE,
(nvl(VOYAGE_AMOUNT,0)+nvl(TAXESandFEES,0)-nvl(VOYAGE_COMMISSION_AMT,0)- CREDIT_CARD_FEE ) AS ACTUAL_NET_TICKET_REVENUE,
(nvl(VOYAGE_AMOUNT,0)+nvl(TAXESandFEES,0)-nvl(VOYAGE_COMMISSION_AMT,0)- CREDIT_CARD_FEE ) AS ACTUAL_NET_NET_TICKET_REVENUE,
(0- nvl(INSURANCE,0)-nvl(VOYAGE_COMMISSION_AMT,0)- CREDIT_CARD_FEE ) AS ACTUAL_NET_OTHER_REVENUE,
(ACTUAL_GROSS_TICKET_REVENUE) AS ACTUAL_GROSS_TOTAL_REVENUE,
(ACTUAL_NET_TICKET_REVENUE+ACTUAL_NET_OTHER_REVENUE) AS ACTUAL_NET_TOTAL_REVENUE,
(ACTUAL_NET_NET_TICKET_REVENUE) AS ACTUAL_NET_NET_TOTAL_REVENUE,
NET_DUE,
currency,
currency_rate
FROM  
(
SELECT snapshot_date,res_id,guest_id,package_id,ship_id,agency_id,sail_id,FACT.invoice_item_type_id,FACT.price_area_id,addon_id,promotion_id,
SUM(amount) as amount,
SUM (CASE WHEN INVOICE_ITEM_TYPE =  'VOYAGE FARE' AND  INVOICE_ITEM_SUBTYPE IN ('SINGLE','VOYAGE FARE DBL','VOYAGE FARE 3-8') THEN AMOUNT END ) AS VOYAGE_AMOUNT,
SUM(CASE WHEN INVOICE_ITEM_TYPE = 'TAXES & FEES' THEN AMOUNT END) AS TAXESandFEES,
--SUM(CASE WHEN INVOICE_ITEM_TYPE = 'TAXES & FEES' AND INVOICE_ITEM_SUBTYPE ='GOVERNMENT TAX' THEN AMOUNT END) AS GOVERNMENT_TAX,
SUM(CASE WHEN PRICE_AREA = 'MANUAL ADJ' THEN AMOUNT END ) AS MANUAL_ADJ,
SUM(CASE WHEN INVOICE_ITEM_TYPE = 'INSURANCE' THEN AMOUNT END) AS VOYAGE_PROTECT,
NET_DUE,
COMMISSION_PERCENT,
currency,
currency_rate 
FROM hive_schema_stg.seaware_revenue_fact FACT 
INNER JOIN seaware.seaware_invoice_item_type_lkp INV on FACT.invoice_item_type_id = INV.invoice_item_type_id 
INNER JOIN seaware.seaware_price_area_lkp PA on FACT.price_area_id = PA.price_area_id  
--where snapshot_date=  (SELECT MAX(snapshot_date) as max_tstamp FROM hive_schema_stg.seaware_revenue_fact)
GROUP BY snapshot_date,res_id,guest_id,package_id,ship_id,agency_id,sail_id,FACT.invoice_item_type_id,FACT.price_area_id,addon_id,promotion_id,COMMISSION_PERCENT,currency,currency_rate,NET_DUE
) BASE_SQL

with no schema binding;
