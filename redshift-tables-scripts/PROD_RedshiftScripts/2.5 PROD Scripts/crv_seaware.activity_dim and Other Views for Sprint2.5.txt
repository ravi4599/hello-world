
create or replace view seaware.seaware_addon_fact
as select * from hive_schema_stg.seaware_addon_fact
with no schema binding;


CREATE OR REPLACE VIEW seaware.seaware_addon_invoice
AS 
SELECT 
   snapshot_date,             
   res_id,                          
   guest_id,
   addon_id,    
   invoice_item_type_id,            
   price_area_id,                   
   amount,                       
   currency,                
   currency_rate                 
FROM hive_schema_stg.seaware_revenue_fact
WHERE addon_id <> -1
and snapshot_date = ( Select max(snapshot_date) from hive_schema_stg.seaware_revenue_fact)
with no schema binding;
;
