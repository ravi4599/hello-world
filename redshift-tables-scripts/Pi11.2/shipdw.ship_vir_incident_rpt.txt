CREATE TABLE IF NOT EXISTS shipdw.ship_vir_incident_rpt
(
	incident_id varchar(256)
	,incident_type VARCHAR(256) 
	,incident_category VARCHAR(256)
	,VIR_number VARCHAR(256)
	,priority VARCHAR(256)
	,severity VARCHAR(256)
	,status VARCHAR(256)
	,incident_opened_datetime TIMESTAMP WITHOUT TIME ZONE 
	,incident_opened_datetime_utc TIMESTAMP WITHOUT TIME ZONE 
	,total_resolution_time INTEGER 
	,reported_by VARCHAR(150)
	,assigned_crew VARCHAR(150) 
	,impacted_sailor VARCHAR(150) 
	,sailor_notes VARCHAR(256)  
	,crew_notes VARCHAR(256) 
	,resolution_value_amount FLOAT8 
	,incident_department VARCHAR(256)
	,cabin_number VARCHAR(256)  
	,etl_ld_dt TIMESTAMP WITHOUT TIME ZONE  
)
DISTSTYLE AUTO
;
