CREATE TABLE `DependencyFlag` (
  `SrcTriggeringTool` varchar(10) NOT NULL DEFAULT '',
  `TgtTriggeredTool` varchar(10) NOT NULL DEFAULT '',
  `JobName` varchar(255) NOT NULL DEFAULT '',
  `JobIndicator` tinyint(1) DEFAULT NULL,
  `SrcTriggeringToolUpdTime` timestamp NULL DEFAULT NULL,
  `TgtTriggeredToolUpdTime` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`SrcTriggeringTool`,`TgtTriggeredTool`,`JobName`)
) ENGINE=InnoDB DEFAULT CHARSET=latin1

CREATE TABLE `talend_spark_dependency` (
  `Load_frequency` varchar(40) NOT NULL,
  `Talend_Start` timestamp NULL DEFAULT NULL,
  `Talend_End` timestamp NULL DEFAULT NULL,
  `loadtime` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`Load_frequency`),
)
