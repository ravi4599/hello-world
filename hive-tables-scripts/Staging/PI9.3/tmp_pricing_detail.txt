CREATE VIEW `vv_db`.`tmp_pricing_detail`(sail_id, price_code,max_available_capacity,num_of_adults_passed, guestseq, amount, category) AS
Select src_sail_id as sail_id,price_code,max_available_capacity,num_of_adults_passed,guestseq,amount,category from vv_db.hvtb_nbx_core_pricing_detail where currency='USD'
and  num_of_adults_passed==max_available_capacity
